## Was wir bauen

Ein Node.js-Skript, das Markdown-Dateien aus der Astro Content Collection automatisch übersetzt:

- **Primär:** Deutsch → Englisch (produktiv nutzbar, realer Workshop-Output)
- **Spaß-Zugabe:** Deutsch → Klingonisch oder Elbisch (zeigt, dass das Skript beliebig erweiterbar ist)

Das Skript nutzt den **adesso AI Hub** — einen OpenAI-kompatiblen API-Endpunkt — und lässt sich
mit jedem anderen OpenAI-kompatiblen Dienst (OpenAI, Azure OpenAI, Ollama, etc.) betreiben.

## Voraussetzungen

### adesso AI Hub API Key

Den API Key als Umgebungsvariable setzen — **niemals** direkt in den Code schreiben:

```bash
# Linux/macOS — in ~/.bashrc oder ~/.zshrc
export ADESSO_AI_HUB_API_KEY="dein-api-key-hier"

# Windows (PowerShell)
$env:ADESSO_AI_HUB_API_KEY = "dein-api-key-hier"
```

Für GitHub Actions als Repository Secret hinterlegen:
1. Im Repository auf **Settings** → **Secrets and variables** → **Actions** navigieren
2. **New repository secret** klicken, Name: `ADESSO_AI_HUB_API_KEY`

### Abhängigkeiten installieren

```bash
npm install openai gray-matter
```

| Paket        | Zweck                                               |
| ------------ | --------------------------------------------------- |
| `openai`     | Offizieller OpenAI-Client (kompatibel mit adesso AI Hub) |
| `gray-matter`| Frontmatter aus Markdown-Dateien parsen und schreiben |

## Das Übersetzungs-Skript

### `scripts/translate.mjs`

```javascript
import OpenAI from "openai";
import matter from "gray-matter";
import fs from "fs";
import path from "path";
import { fileURLToPath } from "url";

const __dirname = path.dirname(fileURLToPath(import.meta.url));

// Konfiguration
const CONFIG = {
  apiKey: process.env.ADESSO_AI_HUB_API_KEY,
  baseURL: "https://ai-hub.adesso.de/v1",  // adesso AI Hub Endpunkt
  model: "gpt-4o",
  sourceDir: path.join(__dirname, "../src/content/docs"),
  targetLang: process.argv[2] || "english",
};

// Sprachkonfiguration
const LANGUAGES = {
  english: {
    prompt: "Translate the following German Markdown content to English. " +
      "Preserve all Markdown formatting, code blocks, and frontmatter values exactly. " +
      "Only translate the human-readable text.",
    suffix: "-en",
  },
  klingon: {
    prompt: "Translate the following German Markdown content to Klingon (tlhIngan Hol). " +
      "Preserve all Markdown formatting and code blocks exactly. " +
      "Use authentic Klingon vocabulary and grammar. " +
      "For technical terms without Klingon equivalents, keep the original term.",
    suffix: "-tlh",
  },
  elvish: {
    prompt: "Translate the following German Markdown content to Quenya (High Elvish from Tolkien). " +
      "Preserve all Markdown formatting and code blocks exactly. " +
      "Use authentic Quenya vocabulary. " +
      "For technical terms without Quenya equivalents, keep the original term.",
    suffix: "-qya",
  },
};

// OpenAI-Client initialisieren
const client = new OpenAI({
  apiKey: CONFIG.apiKey,
  baseURL: CONFIG.baseURL,
});

/**
 * Übersetzt einen Markdown-Text über die KI-API
 */
async function translateContent(content, language) {
  const langConfig = LANGUAGES[language];
  if (!langConfig) {
    throw new Error(`Unbekannte Sprache: ${language}. Verfügbar: ${Object.keys(LANGUAGES).join(", ")}`);
  }

  const response = await client.chat.completions.create({
    model: CONFIG.model,
    messages: [
      {
        role: "system",
        content: langConfig.prompt,
      },
      {
        role: "user",
        content: content,
      },
    ],
    temperature: 0.3,   // Niedrig für konsistente, präzise Übersetzungen
  });

  return response.choices[0].message.content;
}

/**
 * Verarbeitet eine einzelne Markdown-Datei
 */
async function translateFile(filePath, language) {
  const langConfig = LANGUAGES[language];
  const raw = fs.readFileSync(filePath, "utf-8");
  const { data: frontmatter, content } = matter(raw);

  console.log(`  Übersetze: ${path.basename(filePath)} → ${language}...`);

  // Nur den Body übersetzen, Frontmatter-Werte separat
  const translatedBody = await translateContent(content, language);

  // Frontmatter-Felder übersetzen (title, description)
  if (frontmatter.title) {
    frontmatter.title = await translateContent(frontmatter.title, language);
  }
  if (frontmatter.description) {
    frontmatter.description = await translateContent(frontmatter.description, language);
  }

  // Ausgabedatei zusammenbauen
  const translatedFile = matter.stringify(translatedBody, frontmatter);

  // Ausgabepfad: gleiches Verzeichnis, Suffix vor der Erweiterung
  const ext = path.extname(filePath);
  const base = path.basename(filePath, ext);
  const dir = path.dirname(filePath);
  const outputPath = path.join(dir, `${base}${langConfig.suffix}${ext}`);

  fs.writeFileSync(outputPath, translatedFile, "utf-8");
  console.log(`  ✓ Gespeichert: ${path.basename(outputPath)}`);

  return outputPath;
}

/**
 * Verarbeitet alle Markdown-Dateien in einem Verzeichnis
 */
async function translateDirectory(dirPath, language) {
  const entries = fs.readdirSync(dirPath, { withFileTypes: true });

  for (const entry of entries) {
    const fullPath = path.join(dirPath, entry.name);

    if (entry.isDirectory()) {
      await translateDirectory(fullPath, language);
    } else if (
      (entry.name.endsWith(".md") || entry.name.endsWith(".mdx")) &&
      !entry.name.includes("-en.") &&
      !entry.name.includes("-tlh.") &&
      !entry.name.includes("-qya.")
    ) {
      await translateFile(fullPath, language);
    }
  }
}

// Hauptprogramm
async function main() {
  if (!CONFIG.apiKey) {
    console.error("❌ ADESSO_AI_HUB_API_KEY ist nicht gesetzt!");
    process.exit(1);
  }

  const language = CONFIG.targetLang;
  console.log(`\n🌍 Starte Übersetzung nach: ${language}`);
  console.log(`📁 Quellverzeichnis: ${CONFIG.sourceDir}\n`);

  await translateDirectory(CONFIG.sourceDir, language);

  console.log("\n✅ Übersetzung abgeschlossen!");
}

main().catch((err) => {
  console.error("❌ Fehler:", err.message);
  process.exit(1);
});
```

## npm-Scripts einrichten

In `package.json`:

```json
{
  "scripts": {
    "translate:en": "node scripts/translate.mjs english",
    "translate:klingon": "node scripts/translate.mjs klingon",
    "translate:elvish": "node scripts/translate.mjs elvish"
  }
}
```

## Das Skript ausführen

```bash
# Deutsch → Englisch (produktiv)
npm run translate:en

# Deutsch → Klingonisch (Spaß)
npm run translate:klingon

# Deutsch → Elbisch/Quenya (Spaß)
npm run translate:elvish
```

## Prompt-Engineering: Warum `temperature: 0.3`?

Der `temperature`-Parameter steuert die Kreativität der KI:

| Wert  | Verhalten                                                  | Einsatz              |
| ----- | ---------------------------------------------------------- | -------------------- |
| `0.0` | Deterministisch, immer gleiche Ausgabe                     | Technische Übersetzung |
| `0.3` | Konsistent, minimale Variation — **unser Wert**            | Dokumentation        |
| `0.7` | Kreativ, variiert bei gleicher Eingabe                     | Texterstellung       |
| `1.0` | Sehr kreativ, manchmal unvorhersehbar                      | Brainstorming        |

Für Klingonisch und Elbisch kann `temperature: 0.5` interessantere Ergebnisse liefern — die KI
darf hier etwas kreativer sein.

## Qualität der Übersetzungen verbessern

### System-Prompt verfeinern

Für bessere Englisch-Übersetzungen den Prompt erweitern:

```javascript
english: {
  prompt: `You are a professional technical writer and translator.
Translate the following German Markdown documentation to English.

Rules:
- Preserve ALL Markdown formatting (headers, lists, code blocks, links)
- Keep ALL code examples and commands unchanged
- Translate only human-readable text
- Use clear, concise technical English (Microsoft Writing Style Guide)
- Keep German proper nouns (company names, product names) unchanged
- Translate UI elements and button labels consistently`,
  suffix: "-en",
},
```

### Glossar für konsistente Begriffe

Wiederkehrende Fachbegriffe im Prompt festlegen:

```javascript
const GLOSSARY = `
Glossary (always use these translations):
- Schulung → Training
- Teilnehmende → Participants
- Linting-Stack → Linting Stack
- Bereitstellung → Deployment
`;

// Im System-Prompt anhängen
content: langConfig.prompt + "\n\n" + GLOSSARY,
```

## Integration in GitHub Actions (optional)

Für automatische Übersetzung bei jedem Release:

```yaml
- name: Translate documentation
  env:
    ADESSO_AI_HUB_API_KEY: ${{ secrets.ADESSO_AI_HUB_API_KEY }}
  run: npm run translate:en
```

> **Hinweis:** KI-Übersetzungen kosten API-Credits. Für GitHub Actions empfiehlt sich,
> den Schritt nur bei explizitem Trigger (`workflow_dispatch`) oder auf Release-Branches
> auszuführen — nicht bei jedem Push.
